# Revenue calculation model
def calculate_revenue(cost):
    if cost > 20:
        return cost * 0.25
    elif cost > 5:
        return cost * 0.15
    else:
        return 0

foodhub['platform_revenue'] = foodhub['cost_of_the_order'].apply(calculate_revenue)
total_revenue = foodhub['platform_revenue'].sum()

print(f"Total Platform Revenue: ${total_revenue:,.2f}")
